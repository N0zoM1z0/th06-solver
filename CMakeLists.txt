cmake_minimum_required(VERSION 3.20)
project(th06_solver VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use clang++: the imported VC7 source contains Microsoft inline assembly")
endif()
add_compile_options(-m32 -fms-extensions -fasm-blocks -fno-strict-aliasing -ffp-contract=off
  -Wno-microsoft-template -Wno-unknown-pragmas -Wno-writable-strings -Wno-address-of-temporary
  -ffunction-sections -fdata-sections)
add_link_options(-m32 -Wl,--gc-sections)
set(native ${CMAKE_CURRENT_SOURCE_DIR}/third_party/th06/src)
configure_file(${native}/i18n.tpl ${CMAKE_CURRENT_BINARY_DIR}/generated/i18n.hpp COPYONLY)
set(units AnmManager AsciiManager BombData BulletData BulletManager EclManager
  EffectManager EnemyEclInstr EnemyManager GameManager Gui ItemManager Player ScreenEffect Stage)
set(game_sources)
foreach(unit IN LISTS units)
  list(APPEND game_sources ${native}/${unit}.cpp)
endforeach()
add_library(th06_resources STATIC src/resources.cpp)
target_include_directories(th06_resources PUBLIC include)
target_compile_features(th06_resources PUBLIC cxx_std_17)
add_library(th06_tape STATIC src/tape.cpp)
target_include_directories(th06_tape PUBLIC include)
target_compile_features(th06_tape PUBLIC cxx_std_17)
add_library(th06_native STATIC ${game_sources} src/native_support.cpp src/platform.cpp src/session.cpp
  third_party/th08-compat/d3d_null.cpp third_party/th08-compat/d3dx_math.cpp)
target_link_libraries(th06_native PUBLIC th06_resources)
target_compile_features(th06_native PUBLIC cxx_std_17)
target_include_directories(th06_native PUBLIC include ${native} third_party/th08-compat/include
  third_party/th08-compat ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_definitions(th06_native PUBLIC TH06_HEADLESS)
target_compile_options(th06_native PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/include/th06/platform.hpp)
add_executable(th06_headless tools/headless.cpp)
target_link_libraries(th06_headless PRIVATE th06_native)
add_library(th06_planner STATIC src/planner.cpp)
target_link_libraries(th06_planner PUBLIC th06_native)
add_executable(th06_plan tools/plan.cpp)
target_link_libraries(th06_plan PRIVATE th06_planner th06_tape)
add_executable(th06_replay_check tools/replay_check.cpp)
target_link_libraries(th06_replay_check PRIVATE th06_native th06_tape)
add_executable(th06_campaign tools/campaign.cpp)
target_link_libraries(th06_campaign PRIVATE th06_tape)

set(TH06_DATA_DIR "" CACHE PATH "Local directory containing the six private TH06 DAT archives")
if(BUILD_TESTING)
  add_executable(resources_tests tests/resources_tests.cpp)
  target_link_libraries(resources_tests PRIVATE th06_resources)
  add_test(NAME resource_contracts COMMAND resources_tests ${CMAKE_CURRENT_BINARY_DIR}/resource-fixture)
  add_executable(tape_tests tests/tape_tests.cpp)
  target_link_libraries(tape_tests PRIVATE th06_tape)
  add_test(NAME tape_contracts COMMAND tape_tests ${CMAKE_CURRENT_BINARY_DIR}/tape-fixture.actions)
  add_executable(native_tests tests/native_tests.cpp)
  target_link_libraries(native_tests PRIVATE th06_native)
  add_test(NAME native_contracts COMMAND native_tests)
  add_test(NAME campaign_manifest COMMAND ${CMAKE_COMMAND}
    -DCAMPAIGN=$<TARGET_FILE:th06_campaign> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/campaign_manifest.cmake)
  add_test(NAME campaign_snapshot COMMAND ${CMAKE_COMMAND}
    -DCAMPAIGN=$<TARGET_FILE:th06_campaign> -DWORK=${CMAKE_CURRENT_BINARY_DIR}/snapshot-fixture
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/campaign_snapshot.cmake)
  if(TH06_DATA_DIR)
    add_executable(story_transition_tests tests/story_transition.cpp)
    target_link_libraries(story_transition_tests PRIVATE th06_native)
    add_test(NAME story_transition_contracts COMMAND story_transition_tests ${TH06_DATA_DIR})
    add_test(NAME extra_bootstrap_contracts COMMAND story_transition_tests ${TH06_DATA_DIR} extra)
    add_test(NAME story_terminal_contracts COMMAND story_transition_tests ${TH06_DATA_DIR} premature)
    add_test(NAME real_data_contracts COMMAND ${CMAKE_COMMAND}
      -DHEADLESS=$<TARGET_FILE:th06_headless> -DPLAN=$<TARGET_FILE:th06_plan>
      -DCHECK=$<TARGET_FILE:th06_replay_check> -DDATA=${TH06_DATA_DIR}
      -DWORK=${CMAKE_CURRENT_BINARY_DIR}/real-data-tests -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/real_data.cmake)
    set_tests_properties(real_data_contracts PROPERTIES TIMEOUT 180)
  endif()
endif()
